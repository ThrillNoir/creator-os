# Shared UI components

Framework: React 19 / Next.js 16 App Router. Components are custom BoldKit-styled primitives using Tailwind v4 utility classes, Radix UI, CVA and Lucide icons.

```tsx
      },
      size: {
        default: "h-11 px-5 py-2 has-[>svg]:px-4",

        "icon-lg": "size-12",
      },
    },
    defaultVariants: {
      variant: "default",
      size: "default",
    },
  }
)

function Button({
  className,
  variant = "default",
  size = "default",
  asChild = false,
  ...props
}: React.ComponentProps<"button"> &
  VariantProps<typeof buttonVariants> & {
    asChild?: boolean
  }) {
  const Comp = asChild ? Slot.Root : "button"

  return (
    <Comp
      data-slot="button"
      data-variant={variant}
      data-size={size}
      className={cn(buttonVariants({ variant, size, className }))}
      {...props}
    />
  )
}

export { Button, buttonVariants }
```

### `cloudflare/thot-os-next/src/components/theme-toggle.tsx`

```tsx
"use client";

import { Moon, Sun } from "lucide-react";
import { useEffect, useState } from "react";
import { Button } from "@/components/ui/button";

const themeStorageKey = "thot-os-theme";

export function ThemeToggle() {
  const [isDark, setIsDark] = useState(true);

  useEffect(() => {
    let storedTheme: string | null = null;
    try {
      storedTheme = window.localStorage.getItem(themeStorageKey);
    } catch {
      // Storage can be unavailable in restricted browser contexts.
    }
    const nextIsDark = storedTheme !== "light";
    document.documentElement.classList.toggle("dark", nextIsDark);
    requestAnimationFrame(() => setIsDark(nextIsDark));
  }, []);

  function toggleTheme() {
    const nextIsDark = !isDark;
    setIsDark(nextIsDark);
    document.documentElement.classList.toggle("dark", nextIsDark);
    try {
      window.localStorage.setItem(themeStorageKey, nextIsDark ? "dark" : "light");
    } catch {
      // The selected theme still applies for the current page session.
    }
  }

  const label = isDark ? "Use light theme" : "Use dark theme";
  return (
    <Button aria-label={label} onClick={toggleTheme} size="icon-sm" title={label} type="button" variant="outline">
      {isDark ? <Sun aria-hidden="true" /> : <Moon aria-hidden="true" />}
    </Button>
  );
}
```
